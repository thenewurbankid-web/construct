'use client';
// @line-base pages/ListPage
// @line-route /picks
import { useState } from 'react';
import { ListPage } from '../../../library/pages/ListPage';
import { Button } from '../../../library/components/Button';
import books from '../../../library/fixtures/books.json';
import { listPicks, removePick } from '../store/picksStore';

type Book = { id: string; title: string; author: string };

const allBooks = books as unknown as Book[];

export function PicksPage() {
  const [ids, setIds] = useState<string[]>(() => listPicks());

  const picked = ids
    .map((id) => allBooks.find((b) => b.id === id))
    .filter((b): b is Book => b !== undefined);

  function handleRemove(id: string) {
    removePick(id);
    setIds(listPicks());
  }

  return (
    <ListPage title="My picks" empty={picked.length === 0 ? <p>No picks yet.</p> : undefined}>
      {picked.map((book) => (
        <article key={book.id}>
          <h2>{book.title}</h2>
          <p>{book.author}</p>
          <Button label={`Remove ${book.title}`} variant="secondary" onClick={() => handleRemove(book.id)} />
        </article>
      ))}
    </ListPage>
  );
}
